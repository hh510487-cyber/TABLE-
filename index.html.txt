/* ============================================================
   منطق الأعمال: تذكيرات واتساب + فترة السماح + عدم الحضور التلقائي
   ============================================================ */
function bookingDateTime(b){
  const [y,m,d] = b.booking_date.split('-').map(Number);
  const [hh,mm] = b.booking_time.split(':').map(Number);
  return new Date(y, m-1, d, hh, mm);
}

function ensureReliability(customerId){
  if(!state.reliability[customerId]){
    state.reliability[customerId] = { total_bookings:0, total_noshows:0, is_restricted:false };
  }
  return state.reliability[customerId];
}

function runAutoChecks(){
  const now = state.simTime;
  state.bookings.forEach(b=>{
    if(b.status==='cancelled') return;
    const dt = bookingDateTime(b);
    const reminderAt = addMinutes(dt, -REMINDER_LEAD_MINUTES);
    const graceEnd = addMinutes(dt, GRACE_MINUTES);

    // 1) إرسال التذكير قبل الموعد بساعة بالضبط (أو ما بعده إذا فات الوقت بالمحاكاة)
    if(!b.reminder_sent_at && now >= reminderAt && (b.status==='pending' || b.status==='confirmed')){
      b.reminder_sent_at = now;
    }
    // 2) بعد فترة السماح: تحويل الحجوزات غير المؤكدة إلى "لم يحضر" تلقائيًا وتحرير الطاولة
    if(b.status==='pending' && now >= graceEnd){
      b.status = 'noshow';
      const rel = ensureReliability(b.customer_id);
      rel.total_noshows += 1;
      if(rel.total_noshows >= 3) rel.is_restricted = true;
    }
  });
}

function createBooking({restaurant_id, customer_id, date, time, party_size}){
  const b = {
    id:uid('bk'), code:bookingCode(), restaurant_id, customer_id,
    booking_date:date, booking_time:time, party_size:Number(party_size),
    status:'pending', reminder_sent_at:null, confirmed_at:null,
    created_at:new Date(state.simTime),
  };
  state.bookings.push(b);
  const rel = ensureReliability(customer_id);
  rel.total_bookings += 1;
  return b;
}

const AR_DAYS = ['الأحد','الاثنين','الثلاثاء','الأربعاء','الخميس','الجمعة','السبت'];
const AR_MONTHS = ['يناير','فبراير','مارس','أبريل','مايو','يونيو','يوليو','أغسطس','سبتمبر','أكتوبر','نوفمبر','ديسمبر'];
function fmtDate(dateStr){
  const [y,m,d] = dateStr.split('-').map(Number);
  const dt = new Date(y,m-1,d);
  return `${AR_DAYS[dt.getDay()]} ${d} ${AR_MONTHS[m-1]}`;
}
function fmtTime(t){
  const [hh,mm] = t.split(':').map(Number);
  const period = hh>=12 ? 'م' : 'ص';
  let h12 = hh % 12; if(h12===0) h12=12;
  return `${h12}:${String(mm).padStart(2,'0')} ${period}`;
}
function fmtClock(dt){
  const hh = dt.getHours(), mm = dt.getMinutes();
  const period = hh>=12 ? 'م':'ص';
  let h12 = hh%12; if(h12===0) h12=12;
  return `${AR_DAYS[dt.getDay()]} ${String(h12).padStart(2,'0')}:${String(mm).padStart(2,'0')} ${period}`;
}

function otpAutoNext(el){
  el.value = el.value.replace(/[^0-9]/g,'');
  if(el.value && el.nextElementSibling) el.nextElementSibling.focus();
}
function submitOtp(){
  const digits = Array.from(document.querySelectorAll('.otp-digit')).map(i=>i.value).join('');
  if(digits.length<4){ toastMsg('أكمل الرمز'); render(); return; }
  if(digits !== state.otpPending.code){ toastMsg('رمز غير صحيح، حاول مجددًا'); render(); return; }
  const phone = state.otpPending.phone;
  let cust = state.customers.find(c=>c.phone===phone);
  state.otpPending = null;
  if(cust){
    state.currentCustomer = cust;
    goto('c-home');
  } else {
    state.customerFlow.phone = phone;
    goto('c-name');
  }
}

function ownerSetStatus(id, status){
  const b = state.bookings.find(x=>x.id===id);
  b.status = status;
  if(status==='confirmed') b.confirmed_at = new Date(state.simTime);
  if(status==='noshow'){
    const rel = ensureReliability(b.customer_id);
    rel.total_noshows += 1;
    if(rel.total_noshows>=3) rel.is_restricted = true;
  }
  render();
}

function adminToggleTier(id){
  const r = restaurantById(id);
  r.tier = r.tier==='founding' ? 'standard' : 'founding';
  render();
}
function adminToggleActive(id){
  const r = restaurantById(id);
  r.is_active = !r.is_active;
  render();
}
function adminDeleteRestaurant(id){
  state.restaurants = state.restaurants.filter(r=>r.id!==id);
  state.bookings = state.bookings.filter(b=>b.restaurant_id!==id);
  render();
}
function adminAddRestaurant(){
  const name = document.getElementById('newRName').value.trim();
  const phone = document.getElementById('newRPhone').value.trim();
  const tier = document.getElementById('newRTier').value;
  if(!name||!phone){ toastMsg('أكمل بيانات المطعم'); render(); return; }
  state.restaurants.push({ id:uid('r'), name, phone, tier, is_active:true });
  toastMsg('تمت إضافة المطعم');
  render();
}

/* ============================================================
   إشعارات صغيرة (toast مبسّط عبر alert بديل غير مزعج)
   ============================================================ */
function toastMsg(msg){
  state.toast = msg;
  // إظهار بسيط أعلى الشاشة لمدة قصيرة
  clearTimeout(window.__toastTimer);
  window.__toastTimer = setTimeout(()=>{ state.toast=null; },2500);
  setTimeout(showToastBar,0);
}
function showToastBar(){
  let bar = document.getElementById('toastBar');
  if(bar) bar.remove();
  if(!state.toast) return;
  bar = document.createElement('div');
  bar.id='toastBar';
  bar.style.cssText = 'position:fixed;top:14px;left:50%;transform:translateX(-50%);background:var(--ink);color:#fff;padding:10px 18px;border-radius:999px;font-size:13px;z-index:50;box-shadow:0 8px 20px rgba(0,0,0,.25);';
  bar.textContent = state.toast;
  document.body.appendChild(bar);
  setTimeout(()=>{ if(bar) bar.remove(); }, 2400);
}
