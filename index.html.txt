function toast(msg){
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.add('show');
  clearTimeout(t._timer);
  t._timer = setTimeout(()=>t.classList.remove('show'), 2600);
}

function switchRole(role){
  document.querySelectorAll('nav.roles button').forEach(b=>b.classList.toggle('active', b.dataset.role===role));
  document.querySelectorAll('main .view').forEach(v=>v.classList.remove('active'));
  document.getElementById('view-'+role).classList.add('active');
  if(role==='owner') populateOwnerSelect();
  if(role==='admin') renderAdmin();
}

function showStep(id){
  ['cust-step-phone','cust-step-otp','cust-step-name','cust-step-main','cust-step-book','cust-step-success']
    .forEach(s=>document.getElementById(s).style.display = (s===id?'block':'none'));
}

function fmtDate(d){
  const dt = new Date(d+'T00:00:00');
  return dt.toLocaleDateString('ar-IQ',{weekday:'long', day:'numeric', month:'long'});
}

/* ============================================================
   واجهة الزبون
   ============================================================ */
document.querySelectorAll('.otp-d').forEach((inp,idx,arr)=>{
  inp.addEventListener('input', ()=>{
    if(inp.value && idx<arr.length-1) arr[idx+1].focus();
  });
});

function sendOTP(){
  const phone = document.getElementById('cust-phone-input').value.trim();
  if(phone.length < 8){ toast('أدخل رقم هاتف صحيح'); return; }
  session.pendingPhone = phone;
  session.pendingOTP = String(Math.floor(1000 + Math.random()*9000));
  document.getElementById('otp-sent-to').textContent = 'أرسلنا رمز التحقق إلى ' + phone;
  document.getElementById('otp-hint').textContent = session.pendingOTP;
  document.querySelectorAll('.otp-d').forEach(i=>i.value='');
  showStep('cust-step-otp');
}

function verifyOTP(){
  const entered = Array.from(document.querySelectorAll('.otp-d')).map(i=>i.value).join('');
  if(entered.length<4){ toast('أدخل الرمز كاملًا'); return; }
  if(entered !== session.pendingOTP){ toast('رمز غير صحيح، حاول مرة أخرى'); return; }

  const phone = session.pendingPhone;
  session.custPhone = phone;

  if(!customers[phone]){
    showStep('cust-step-name');
  } else {
    showStep('cust-step-main');
    renderCustomerHome();
  }
}

function renderCustomerHome(){
  const c = customers[session.custPhone];
  document.getElementById('cust-welcome-name').textContent = c.name;

  const box = document.getElementById('cust-restriction-box');
  box.innerHTML = c.restricted
    ? `<div class="card restrict-flag">⚠️ حسابك مقيّد مؤقتًا بسبب تكرار عدم الحضور (${c.noshowCount} مرات). يرجى التواصل مع المطعم مباشرة لإتمام الحجز.</div>`
    : (c.noshowCount>=2 ? `<div class="card restrict-flag">تنبيه: لديك ${c.noshowCount} حالة عدم حضور سابقة. حالة إضافية قد تؤدي لتقييد حسابك.</div>` : '');

  const list = document.getElementById('cust-rest-list');
  list.innerHTML = restaurants.filter(r=>r.active).map(r=>`
    <div class="rest-item" onclick="openBooking('${r.id}')">
      <h3>${r.name}</h3>
      <span class="badge ${r.tier==='founding'?'badge-founding':'badge-standard'}">${r.tier==='founding'?'مطعم مؤسِّس':'مطعم شريك'}</span>
    </div>
  `).join('') || '<div class="empty">لا توجد مطاعم متاحة حاليًا</div>';

  const mine = bookings.filter(b=>b.customerPhone===session.custPhone).sort((a,b)=> (a.date+a.time < b.date+b.time ? 1:-1));
  const bl = document.getElementById('cust-my-bookings');
  bl.innerHTML = mine.length ? mine.map(b=>{
    const r = restaurants.find(x=>x.id===b.restaurantId);
    return bookingRowHTML(b, r.name, false);
  }).join('') : '<div class="empty">ما عندك حجوزات بعد</div>';
}

function openBooking(restId){
  if(customers[session.custPhone].restricted){ toast('حسابك مقيّد — تواصل مع المطعم مباشرة'); return; }
  session.bookingTargetRestId = restId;
  const r = restaurants.find(x=>x.id===restId);
  document.getElementById('book-rest-name').textContent = r.name;
  document.getElementById('book-date').value = new Date().toISOString().slice(0,10);
  document.getElementById('book-time').value = '19:00';
  document.getElementById('book-party').value = 2;
  document.getElementById('book-large-warning').style.display='none';
  showStep('cust-step-book');
}

document.getElementById('book-party').addEventListener('input', ()=>{
  const v = parseInt(document.getElementById('book-party').value||0);
  document.getElementById('book-large-warning').style.display = v>=6 ? 'block' : 'none';
});

function confirmBooking(){
  const date = document.getElementById('book-date').value;
  const time = document.getElementById('book-time').value;
  const party = parseInt(document.getElementById('book-party').value||0);
  if(!date || !time || !party){ toast('أكمل كل الحقول'); return; }

  bookingCounter++;
  const code = 'B'+bookingCounter;
  const b = {
    id:'bk'+bookingCounter,
    code,
    restaurantId: session.bookingTargetRestId,
    customerPhone: session.custPhone,
    date, time, party,
    status:'pending',
    needsManualCall: party>=6,
  };
  bookings.push(b);
  customers[session.custPhone].bookingsCount++;

  document.getElementById('success-code').textContent = code;
  showStep('cust-step-success');
}

function bookingRowHTML(b, restName, ownerMode){
  const badgeClass = {pending:'badge-pending',confirmed:'badge-confirmed',noshow:'badge-noshow',cancelled:'badge-cancelled'}[b.status];
  const badgeText = {pending:'بانتظار التأكيد',confirmed:'مؤكد',noshow:'لم يحضر',cancelled:'ملغى'}[b.status];
  let actions = '';
  if(ownerMode && b.status==='pending'){
    actions = `<div class="actions">
      <button class="btn btn-primary btn-small" onclick="ownerSetStatus('${b.id}','confirmed')">تأكيد الحضور</button>
      <button class="btn btn-danger btn-small" onclick="ownerSetStatus('${b.id}','noshow')">لم يحضر</button>
    </div>`;
  }
  const callFlag = b.needsManualCall ? `<div class="muted" style="color:var(--wine-2);margin-top:4px;">📞 يتطلب تأكيد هاتفي يدوي (حجز ${b.party} أشخاص)</div>` : '';
  return `<div class="booking-row">
    <div class="top-row">
      <div>
        <div class="code">${b.code}</div>
        <div class="meta">${restName ? restName+' — ' : ''}${fmtDate(b.date)} · ${b.time} · ${b.party} أشخاص</div>
      </div>
      <span class="badge ${badgeClass}">${badgeText}</span>
    </div>
    ${callFlag}
    ${actions}
  </div>`;
}

function renderOwnerDash(){
  const r = restaurants.find(x=>x.id===session.ownerRestId);
  document.getElementById('owner-rest-name').textContent = r.name;

  const mine = bookings.filter(b=>b.restaurantId===r.id);
  const total = mine.length;
  const confirmed = mine.filter(b=>b.status==='confirmed').length;
  const noshow = mine.filter(b=>b.status==='noshow').length;

  document.getElementById('owner-stats').innerHTML = `
    <div class="stat"><span class="num">${total}</span><span class="lbl">إجمالي الحجوزات</span></div>
    <div class="stat"><span class="num">${confirmed}</span><span class="lbl">مؤكدة</span></div>
    <div class="stat"><span class="num">${noshow}</span><span class="lbl">لم يحضر</span></div>
  `;

  const upcoming = mine.filter(b=>b.status==='pending' || b.status==='confirmed')
                        .sort((a,b)=> (a.date+a.time > b.date+b.time ? 1:-1));
  document.getElementById('owner-bookings').innerHTML = upcoming.length
    ? upcoming.map(b=>bookingRowHTML(b, null, true)).join('')
    : '<div class="empty">لا توجد حجوزات قادمة</div>';
}

function ownerSetStatus(bookingId, status){
  const b = bookings.find(x=>x.id===bookingId);
  b.status = status;
  if(status==='confirmed') b.confirmedAt = new Date().toISOString();

  if(status==='noshow'){
    const c = customers[b.customerPhone];
    if(c){
      c.noshowCount++;
      if(c.noshowCount>=3) c.restricted = true;
    }
  }
  toast(status==='confirmed' ? 'تم تأكيد الحضور' : 'تم تسجيل عدم الحضور');
  renderOwnerDash();
}

function addRestaurant(){
  const name = document.getElementById('admin-new-name').value.trim();
  const phone = document.getElementById('admin-new-phone').value.trim();
  if(!name || !phone){ toast('أكمل بيانات المطعم'); return; }
  const id = 'r'+(restaurants.length+1)+Math.floor(Math.random()*1000);
  restaurants.push({id, name, phone, tier:newTier, active:true});
  document.getElementById('admin-new-name').value='';
  document.getElementById('admin-new-phone').value='';
  toast('تمت إضافة المطعم');
  renderAdmin();
}

function toggleRestActive(id){
  const r = restaurants.find(x=>x.id===id);
  r.active = !r.active;
  renderAdmin();
}

function changeRestTier(id, tier){
  const r = restaurants.find(x=>x.id===id);

  renderAdmin();
}

function removeRestaurant(id){
  restaurants = restaurants.filter(x=>x.id!==id);

}
